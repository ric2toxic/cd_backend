CREATE TABLE `oc_courier_partners` ( `id` INT(10) NOT NULL A UTO_INCREMENT COMMENT 'id of courier company' , `courier_name`VARCHAR(100) CHARACTER SET utf8 COLLATE utf8_general_ci NOT NULL COMMENT 'name of the courier company' , `courier_telephone`VARCHAR(10) CHARACTER SET utf8 COLLATE utf8_general_ci NULL DEFAULT NULL COMMENT 'telephone of courier company' , PRIMARY KEY (`id`))ENGINE = MyISAM CHARACTER SET utf8 COLLATE utf8_general_ci COMMENT ='courier partners';
