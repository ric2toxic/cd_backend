Query for trigger_update_seller_status

DROP TRIGGER IF EXISTS `trigger_update_seller_status`;CREATE DEFINER=`root`@`localhost` TRIGGER `trigger_update_seller_status` AFTER UPDATE ON `oc_ms_seller` FOR EACH ROW BEGIN
 IF NEW.seller_status <> OLD.seller_status THEN
    UPDATE `oc_product` p
    SET p.date_modified = NOW()
    WHERE p.product_id IN 
    (SELECT omp.product_id 
     FROM `oc_ms_product` omp 
     WHERE omp.seller_id = OLD.seller_id);
 END IF;
END
