CREATE TABLE `wholesalebox`.`oc_user_changelog` ( `log_id` INT(11) NOT NULL , `user_id` INT(11) NOT NULL , `username` VARCHAR(20) CHARACTER SET utf8 COLLATE utf8_general_ci NOT NULL , `change_type` ENUM('order_apply_credit','customer_add_credit','order_edit_shipping','order_apply_advance') CHARACTER SET utf8 COLLATE utf8_general_ci NOT NULL , `comment` VARCHAR(255) CHARACTER SET utf8 COLLATE utf8_general_ci NOT NULL , `date_added` DATETIME NOT NULL , PRIMARY KEY (`log_id`)) ENGINE = MyISAM CHARACTER SET utf8 COLLATE utf8_general_ci COMMENT = 'Table to log critical admin panel edits done by various user';
